function openModal(index){
  currentProductIndex = index;
  const p = products[index];
  prodTitle.textContent = p.name;
  prodDesc.textContent = p.desc;
  mainImg.src = p.images[0];
  buyNow.href = p.wa;
  buyNow.setAttribute('aria-label', `Buy ${p.name} on WhatsApp`);

  // thumbnails
  thumbsCol.innerHTML = '';
  p.images.forEach((src,i)=>{
    const img = document.createElement('img');
    img.src = src;
    img.alt = `${p.name} thumb ${i+1}`;
    if(i===0) img.classList.add('active');
    img.addEventListener('click', ()=> {
      mainImg.src = src;
      thumbsCol.querySelectorAll('img').forEach(x=>x.classList.remove('active'));
      img.classList.add('active');
    });
    thumbsCol.appendChild(img);
  });

  // sizes chips
  sizesRow.innerHTML = '';
  p.sizes.forEach((s,i)=>{
    const label = document.createElement('div');
    label.className = 'chipLabel';
    label.textContent = s;
    label.dataset.value = s;
    label.addEventListener('click', ()=> {
      sizesRow.querySelectorAll('.chipLabel').forEach(x=>x.classList.remove('selected'));
      label.classList.add('selected');
      currentSize = s;
    });
    sizesRow.appendChild(label);
    if(i===0){ label.classList.add('selected'); currentSize = s; }
  });

  // colors
  colorsRow.innerHTML = '';
  p.colors.forEach((c,i)=>{
    const dot = document.createElement('div');
    dot.className = 'colorDot';
    dot.style.background = c;
    dot.dataset.value = c;
    dot.addEventListener('click', ()=> {
      colorsRow.querySelectorAll('.colorDot').forEach(x=>x.classList.remove('selected'));
      dot.classList.add('selected');
      currentColor = c;
    });
    colorsRow.appendChild(dot);
    if(i===0){ dot.classList.add('selected'); currentColor = c; }
  });

  // show modal
  backdrop.style.display = 'flex';
  backdrop.setAttribute('aria-hidden','false');
  setTimeout(()=> window.addEventListener('keydown', escHandler), 80);
}

function escHandler(e){ if(e.key === 'Escape') closeModal(); }
function closeModal(){ backdrop.style.display = 'none'; backdrop.setAttribute('aria-hidden','true'); window.removeEventListener('keydown', escHandler); }

/* close handlers */
closeBtn.addEventListener('click', closeModal);
backdrop.addEventListener('click', (ev)=> { if(ev.target === backdrop) closeModal(); });

/* Buy button: append size/color info into WhatsApp text param */
buyNow.addEventListener('click', function(e){
  if(currentProductIndex === null) return;
  const p = products[currentProductIndex];
  const sizeText = currentSize ? `Size: ${currentSize}` : '';
  const colorText = currentColor ? `Color: ${currentColor}` : '';
  const message = encodeURIComponent(`Hi, I'm interested in ${p.name}.\n${sizeText}\n${colorText}`);
  if(buyNow.href && buyNow.href.includes('wa.me')){
    const sep = buyNow.href.includes('?') ? '&' : '?';
    // open in new tab (link has target _blank) so we can update href
    buyNow.href = p.wa + sep + 'text=' + message;
  }
});

/* small utility */
function escapeHtml(str){
  return String(str).replace(/[&<>"']/g, function(m){ return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m]; });
}
